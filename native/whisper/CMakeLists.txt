cmake_minimum_required(VERSION 3.21)
project(spelunkr_captions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86_64|AMD64|amd64)")
  # Load the fastest compatible kernel at runtime, with a baseline fallback.
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
  set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
endif()
if(APPLE)
  set(CMAKE_BUILD_RPATH "@loader_path")
else()
  set(CMAKE_BUILD_RPATH "$ORIGIN")
  set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
endif()
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# Keep the distributed x64 helper usable on older Windows 10 CPUs.
foreach(feature AVX AVX2 AVX512 FMA F16C)
  set(GGML_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${WHISPER_SOURCE} whisper-build EXCLUDE_FROM_ALL)
add_executable(caption-reader main.cpp)
add_executable(caption-transcript-test transcript-test.cpp)
target_link_libraries(caption-reader PRIVATE whisper)
if(WIN32)
  target_compile_definitions(caption-reader PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(caption-reader PRIVATE ole32 uuid avrt)
endif()
